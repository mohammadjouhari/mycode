        function applyFiltersAndSort() {
            const maxPrice = parseFloat(document.getElementById('priceRange').value);
            
            // Checked stops
            const checkedStops = Array.from(document.querySelectorAll('.filter-stop:checked')).map(el => parseInt(el.value));
            
            // Checked airlines
            const checkedAirlines = Array.from(document.querySelectorAll('.filter-airline:checked')).map(el => el.value);

            currentFilteredFlights = sampleFlights.filter(flight => {
                const matchesPrice = flight.price <= maxPrice;
                const matchesStops = checkedStops.includes(flight.stops) || (flight.stops >= 2 && checkedStops.includes(2));
                const matchesAirline = checkedAirlines.includes(flight.airline);
                return matchesPrice && matchesStops && matchesAirline;
            });

            // Sorting
            const sortMode = document.getElementById('sortSelect').value;
            if (sortMode === 'price-asc') {
                currentFilteredFlights.sort((a, b) => a.price - b.price);
            } else if (sortMode === 'price-desc') {
                currentFilteredFlights.sort((a, b) => b.price - a.price);
            } else if (sortMode === 'duration-asc') {
                currentFilteredFlights.sort((a, b) => parseInt(a.duration) - parseInt(b.duration));
            }

            renderFlightResults(currentFilteredFlights);
        }

                             onclick="selectSeat(this, '${seatId}')" id="seat-${seatId}">
                            ${seatId}
                        </div>
                    `;
                });
            }
            container.innerHTML = html;
        }

        function selectSeat(el, seatId) {
            if (el.classList.contains('occupied')) return;

            document.querySelectorAll('.seat.selected').forEach(s => s.classList.remove('selected'));
            el.classList.add('selected');
            selectedSeat = seatId;
        }

        function updatePriceSummary() {
            if (!selectedFlight) return;
            const baseTotal = selectedFlight.price * passengerCount;
            const seatFee = selectedSeat ? 20 : 15;
            const taxes = Math.round(baseTotal * 0.1);
            const total = baseTotal + seatFee + taxes;

            document.getElementById('summaryBasePrice').innerText = `$${baseTotal}`;
            document.getElementById('summarySeatPrice').innerText = `$${seatFee}`;
            document.getElementById('summaryTaxPrice').innerText = `$${taxes}`;
            document.getElementById('summaryTotalPrice').innerText = `$${total}`;
        }

        function completeBooking() {
            // Hide wizard & header controls
            document.getElementById('bookingTabsContent').classList.add('d-none');
            document.getElementById('bookingTabs').classList.add('d-none');

            // Generate Random Booking Ref
            const ref = '#' + Math.random().toString(36).substring(2, 8).toUpperCase();
            document.getElementById('bookingRef').innerText = ref;

            // Show Success Message
            document.getElementById('bookingSuccessMessage').classList.remove('d-none');
        }
    </script>
