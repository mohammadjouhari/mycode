import React, { useState } from 'react';
import DataTable from 'react-data-table-component';

// 1. Define your table columns
const columns = [
  {
    name: 'ID',
    selector: row => row.id,
    sortable: true,
  },
  {
    name: 'Name',
    selector: row => row.name,
    sortable: true,
  },
  {
    name: 'Role',
    selector: row => row.role,
    sortable: true,
  },
];

// 2. Mock some sample data
const initialData = [
  { id: 1, name: 'Alice Smith', role: 'Engineer' },
  { id: 2, name: 'Bob Jones', role: 'Designer' },
  { id: 3, name: 'Charlie Brown', role: 'Product Manager' },
    { id: 4, name: 'Alice Smith', role: 'Engineer' },
  { id: 5, name: 'Bob Jones', role: 'Designer' },
  { id: 6, name: 'Charlie Brown', role: 'Product Manager' },

    { id: 7, name: 'Alice Smith', role: 'Engineer' },
  { id: 8, name: 'Bob Jones', role: 'Designer' },
  { id: 9, name: 'Charlie Brown', role: 'Product Manager' },
    { id: 10, name: 'Alice Smith', role: 'Engineer' },
  { id: 11, name: 'Bob Jones', role: 'Designer' },
  { id: 12, name: 'Charlie Brown', role: 'Product Manager' }
];

export default function MyDataTable() {
  const [data, setData] = useState(initialData);

  // 3. Optional: Search Filter Logic
  const handleFilter = (e) => {
    const value = e.target.value.toLowerCase();
    const filtered = initialData.filter(row => 
      row.name.toLowerCase().includes(value) || 
      row.role.toLowerCase().includes(value)
    );
    setData(filtered);
  };

  return (
    <div style={{ padding: '20px' }}>
      {/* Search Input Bar */}
      <input 
        type="text" 
        placeholder="Search..." 
        onChange={handleFilter} 
        style={{ marginBottom: '15px', padding: '8px', width: '350px' }}
      />
      
      {/* The Data Table Component */}
      <DataTable
        title="Team Directory"
        columns={columns}
        data={data}
        pagination
        highlightOnHover
        pointerOnHover
      />
    </div>
  );
}
