    <script src="https://cdnjs.cloudflare.com/ajax/libs/jquery/3.4.1/jquery.min.js" crossorigin="anonymous"></script>

    <script>
      function show_loading_bar() {
        var $loading_bar = $(".neon-loading-bar");
        if ($loading_bar.length === 0) {
          $loading_bar = $('<div class="neon-loading-bar"><span class="bar"></span></div>');
          $("body").append($loading_bar);
        }
        $loading_bar.removeClass('progress-is-hidden');
      }

      function hide_loading_bar() {
        $(".neon-loading-bar").addClass('progress-is-hidden');
      }

      // Show spinner on start
      show_loading_bar();

      // Automatically hide the spinner after 5 seconds (5000 ms)
      setTimeout(function() {
        hide_loading_bar();
      }, 100);
    </script>
