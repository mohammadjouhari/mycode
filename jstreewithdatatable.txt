<script src="https://cdnjs.cloudflare.com/ajax/libs/jstree/3.3.12/jstree.min.js"></script>
<script src="https://cdn.datatables.net/1.13.6/js/jquery.dataTables.min.js"></script>
<script src="https://cdn.datatables.net/1.13.6/js/dataTables.bootstrap5.min.js"></script>

<script>
    // 1. Sample Data Source
    const allProducts = {
        "laptops": [
            { id: 101, name: "Dell XPS 13", category: "Laptops", price: "$1,200" },
            { id: 102, name: "MacBook Pro 16", category: "Laptops", price: "$2,400" },
            { id: 103, name: "Lenovo ThinkPad X1", category: "Laptops", price: "$1,400" }
        ],
        "phones": [
            { id: 201, name: "iPhone 15 Pro", category: "Smartphones", price: "$999" },
            { id: 202, name: "Samsung Galaxy S24", category: "Smartphones", price: "$899" },
            { id: 203, name: "Google Pixel 8", category: "Smartphones", price: "$699" }
        ],
        "monitors": [
            { id: 301, name: "Dell UltraSharp 27\"", category: "Monitors", price: "$450" },
            { id: 302, name: "LG 34\" Ultrawide", category: "Monitors", price: "$750" }
        ]
    };

    $(document.body).ready(function () {

        // 2. Initialize DataTable
        const dataTable = $('#itemsTable').DataTable({
            data: [], // Starts empty
            columns: [
                { data: 'id' },
                { data: 'name' },
                { data: 'category' },
                { data: 'price' }
            ]
        });

        // 3. Initialize jsTree
        $('#categoryTree').jstree({
            'core': {
                'data': [
                    {
                        'id': 'all_products',
                        'text': 'All Products',
                        'state': { 'opened': true },
                        'children': [
                            {
                                'id': 'hardware',
                                'text': 'Hardware',
                                'state': { 'opened': true },
                                'children': [
                                    { 'id': 'laptops', 'text': 'Laptops', 'icon': 'jstree-file' },
                                    { 'id': 'phones', 'text': 'Smartphones', 'icon': 'jstree-file' }
                                ]
                            },
                            {
                                'id': 'peripherals',
                                'text': 'Peripherals',
                                'state': { 'opened': true },
                                'children': [
                                    { 'id': 'monitors', 'text': 'Monitors', 'icon': 'jstree-file' }
                                ]
                            }
                        ]
                    }
                ]
            }
        });

        // 4. Handle jsTree Node Selection
        $('#categoryTree').on("changed.jstree", function (e, data) {
            if (data.selected.length) {
                const selectedNodeId = data.selected[0];
                const nodeText = data.instance.get_node(selectedNodeId).text;

                // Update Table Header Text
                $('#tableHeader').text('Products: ' + nodeText);

                // Fetch matching data
                let selectedData = [];
                if (selectedNodeId === 'all_products' || selectedNodeId === 'hardware' || selectedNodeId === 'peripherals') {
                    // Combine all sub-categories if parent is selected
                    selectedData = Object.values(allProducts).flat();
                } else if (allProducts[selectedNodeId]) {
                    // Load specific category
                    selectedData = allProducts[selectedNodeId];
                }

                // Update DataTable Content
                dataTable.clear();
                dataTable.rows.add(selectedData);
                dataTable.draw();
            }
        });

    });
</script>
