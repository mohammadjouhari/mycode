    <script>
        $(document).ready(function () {
            const itemsPerPage = 4;
            let currentPage = 1;
            let currentFilter = '*';

            // Initialize Isotope
            var $grid = $('#productGrid').isotope({
                itemSelector: '.grid-item',
                layoutMode: 'fitRows',
                percentPosition: true
            });

            function getFilteredItems() {
                if (currentFilter === '*') {
                    return $('.grid-item');
                }
                return $('.grid-item' + currentFilter);
            }

            function applyPagination() {
                var $filteredItems = getFilteredItems();
                var totalItems = $filteredItems.length;
                var totalPages = Math.ceil(totalItems / itemsPerPage) || 1;

                if (currentPage > totalPages) currentPage = totalPages;
                if (currentPage < 1) currentPage = 1;

                var startIndex = (currentPage - 1) * itemsPerPage;
                var endIndex = startIndex + itemsPerPage;

                // Combine category filter with page slice range
                $grid.isotope({
                    filter: function () {
                        var $this = $(this);
                        var matchesFilter = (currentFilter === '*') || $this.is(currentFilter);
                        if (!matchesFilter) return false;

                        var index = $filteredItems.index(this);
                        return index >= startIndex && index < endIndex;
                    }
                });
                renderPaginationControls(totalPages);
            }

            function renderPaginationControls(totalPages) {
                var $pagination = $('#paginationControls');
                $pagination.empty();

                if (totalPages <= 1) return;

                // Previous Button
                var prevClass = currentPage === 1 ? 'disabled' : '';
                $pagination.append(`
                    <li class="page-item ${prevClass}">
                        <a class="page-link" data-page="${currentPage - 1}">Previous</a>
                    </li>
                `);

                // Page Number Buttons
                for (var i = 1; i <= totalPages; i++) {
                    var activeClass = currentPage === i ? 'active' : '';
                    $pagination.append(`
                        <li class="page-item ${activeClass}">
                            <a class="page-link" data-page="${i}">${i}</a>
                        </li>
                    `);
                }

                // Next Button
                var nextClass = currentPage === totalPages ? 'disabled' : '';
                $pagination.append(`
                    <li class="page-item ${nextClass}">
                        <a class="page-link" data-page="${currentPage + 1}">Next</a>
                    </li>
                `);
            }

            // Left Sidebar Category Event Handler
            $('#categoryFilterPanel').on('click', '.list-group-item', function () {
                currentFilter = $(this).attr('data-filter');
                currentPage = 1;

                $('#categoryFilterPanel .list-group-item').removeClass('active');
                $(this).addClass('active');

                applyPagination();
            });

            // Pagination Click Event Handler
            $('#paginationControls').on('click', 'a.page-link', function (e) {
                e.preventDefault();
                var targetPage = parseInt($(this).attr('data-page'));
                if (targetPage && targetPage !== currentPage) {
                    currentPage = targetPage;
                    applyPagination();
                }
            });

            // Initial Execution
            applyPagination();
        });
    </script>
