  <script>
    // 1. Add a new Group
    function addGroup() {
      const container = document.getElementById("groupContainer");

      const groupCard = document.createElement("div");
      groupCard.className = "group-card";

      groupCard.innerHTML = `
        <div class="d-flex justify-content-between align-items-center mb-3">
          <input type="text" class="form-control form-control-sm font-weight-bold group-name-input" placeholder="Attribute Group Name (e.g. Dimensions)" style="max-width: 70%;">
          <button type="button" class="remove-btn btn-sm" onclick="this.closest('.group-card').remove()">Delete Group</button>
        </div>
        
        <div class="attributes-list"></div>

        <button type="button" class="add-attr-btn btn-sm mt-2" onclick="addAttributeRow(this)">+ Add Key-Value Attribute</button>
      `;

      container.appendChild(groupCard);

      // Add one default attribute row inside the new group
      const addAttrBtn = groupCard.querySelector(".add-attr-btn");
      addAttributeRow(addAttrBtn);
    }

    // 2. Add a Key-Value pair row inside a specific Group
    function addAttributeRow(buttonEl) {
      const attributesList = buttonEl.previousElementSibling;

      const attrRow = document.createElement("div");
      attrRow.className = "attribute-row";

      attrRow.innerHTML = `
        <input type="text" class="form-control form-control-sm attr-key" placeholder="Attribute Key (e.g. Height)">
        <input type="text" class="form-control form-control-sm attr-value" placeholder="Attribute Value (e.g. 10 cm)">
        <button type="button" class="remove-btn btn-sm" onclick="this.closest('.attribute-row').remove()">✕</button>
      `;

      attributesList.appendChild(attrRow);
    }

    // 3. Extract and Serialize all Groups and Key-Value pairs into JSON
    function getAttributesAsJson() {
      const groupCards = document.querySelectorAll(".group-card");
      const result = [];

      groupCards.forEach(card => {
        const groupName = card.querySelector(".group-name-input").value.trim();
        const attrRows = card.querySelectorAll(".attribute-row");
        const attributes = [];

        attrRows.forEach(row => {
          const key = row.querySelector(".attr-key").value.trim();
          const value = row.querySelector(".attr-value").value.trim();

          if (key || value) {
            attributes.push({ key, value });
          }
        });

        result.push({
          groupName: groupName,
          attributes: attributes
        });
      });

      const jsonResult = JSON.stringify(result, null, 2);

      // Log to Browser Console
      console.log("Attribute Groups JSON:", jsonResult);

      // Render on screen
      document.getElementById("jsonOutput").textContent = jsonResult;
    }

    // Event Listeners
    document.getElementById("addGroupBtn").addEventListener("click", addGroup);
    document.getElementById("logJsonBtn").addEventListener("click", getAttributesAsJson);

    // Initialize with 1 default group on load
    addGroup();
  </script>
