  <script>
    // Sample Initial Data Set
    let sampleLoans = [
      { id: "L-1001", name: "Mohammed Al-Hashimi", eid: "784-1988-1092837-1", type: "SHZP Loan", amount: 750000, tenure: 25, status: "Approved" },
      { id: "L-1002", name: "Fatima Salem", eid: "784-1992-4839201-2", type: "EDB Loan", amount: 350000, tenure: 5, status: "Approved" },
      { id: "L-1003", name: "Sultan Al-Zaabi", eid: "784-1985-9920192-3", type: "SHZP Loan", amount: 1200000, tenure: 20, status: "Pending" },
      { id: "L-1004", name: "Rashid Saeed", eid: "784-1995-1029384-4", type: "EDB Loan", amount: 500000, tenure: 10, status: "Rejected" },
      { id: "L-1005", name: "Mariam Al-Kaabi", eid: "784-1991-8839201-5", type: "SHZP Loan", amount: 800000, tenure: 25, status: "Approved" }
    ];

    let dataTableInstance = null;
    const loanModal = new bootstrap.Modal(document.getElementById('loanModal'));

    $(document).ready(function () {
      
      // Initialize DataTable
      dataTableInstance = $('#loanTable').DataTable({
        data: sampleLoans,
        columns: [
          { data: 'id', render: data => `<span class="fw-bold">${data}</span>` },
          { data: 'name' },
          { data: 'eid' },
          { 
            data: 'type',
            render: function(data) {
              const badgeClass = data === 'SHZP Loan' ? 'badge-shzp' : 'badge-edb';
              return `<span class="badge ${badgeClass}">${data}</span>`;
            }
          },
          { 
            data: 'amount', 
            render: data => `AED ${Number(data).toLocaleString()}` 
          },
          { data: 'tenure', render: data => `${data} Yrs` },
          { 
            data: 'status',
            render: function(data) {
              let color = 'bg-secondary';
              if (data === 'Approved') color = 'bg-success';
              if (data === 'Pending') color = 'bg-warning text-dark';
              if (data === 'Rejected') color = 'bg-danger';
              return `<span class="badge ${color}">${data}</span>`;
            }
          },
          {
            data: null,
            orderable: false,
            className: 'text-center',
            render: function(data, type, row) {
              return `
                <button class="btn btn-sm btn-outline-primary action-btn btn-edit" data-id="${row.id}" title="Edit">
                  <i class="bi bi-pencil-square"></i>
                </button>
                <button class="btn btn-sm btn-outline-danger action-btn btn-delete" data-id="${row.id}" title="Delete">
                  <i class="bi bi-trash"></i>
                </button>
              `;
            }
          }
        ],
        order: [[0, 'desc']],
        pageLength: 10
      });

      // Recalculate summary metrics on data updates
      updateMetrics();

      // Dropdown Filter Handlers
      $('#filterLoanType').on('change', function () {
        dataTableInstance.column(3).search(this.value).draw();
      });

      $('#filterStatus').on('change', function () {
        dataTableInstance.column(6).search(this.value).draw();
      });

      // Open Add Modal
      $('#btnOpenAddModal').on('click', function () {
        $('#loanModalTitle').text('New Loan Application');
        $('#loanForm')[0].reset();
        $('#loanId').val('');
        loanModal.show();
      });

      // Open Edit Modal
      $('#loanTable').on('click', '.btn-edit', function () {
        const id = $(this).data('id');
        const rowData = dataTableInstance.rows().data().toArray().find(l => l.id === id);

        if (rowData) {
          $('#loanModalTitle').text(`Edit Loan (${rowData.id})`);
          $('#loanId').val(rowData.id);
          $('#applicantName').val(rowData.name);
          $('#emiratesId').val(rowData.eid);
          $('#loanType').val(rowData.type);
          $('#amount').val(rowData.amount);
          $('#tenure').val(rowData.tenure);
          $('#status').val(rowData.status);
          loanModal.show();
        }
      });

      // Delete Record
      $('#loanTable').on('click', '.btn-delete', function () {
        const id = $(this).data('id');
        if (confirm(`Are you sure you want to delete loan ${id}?`)) {
          const row = dataTableInstance.row($(this).parents('tr'));
          row.remove().draw(false);
          updateMetrics();
        }
      });

      // Handle Form Submission (Add or Update)
      $('#loanForm').on('submit', function (e) {
        e.preventDefault();

        const id = $('#loanId').val();
        const loanPayload = {
          id: id ? id : `L-${Math.floor(1000 + Math.random() * 9000)}`,
          name: $('#applicantName').val(),
          eid: $('#emiratesId').val(),
          type: $('#loanType').val(),
          amount: parseFloat($('#amount').val()),
          tenure: parseInt($('#tenure').val()),
          status: $('#status').val()
        };

        if (id) {
          // Update existing row
          const rowDataIndex = dataTableInstance.rows().data().toArray().findIndex(l => l.id === id);
          dataTableInstance.row(rowDataIndex).data(loanPayload).draw(false);
        } else {
          // Add new row
          dataTableInstance.row.add(loanPayload).draw(false);
        }

        updateMetrics();
        loanModal.hide();
      });

    });

    // Helper: Update Dashboard Cards
    function updateMetrics() {
      const allData = dataTableInstance.rows().data().toArray();
      const totalCount = allData.length;
      const shzpCount = allData.filter(d => d.type === 'SHZP Loan').length;
      const edbCount = allData.filter(d => d.type === 'EDB Loan').length;
      const totalApprovedAmount = allData
        .filter(d => d.status === 'Approved')
        .reduce((sum, d) => sum + Number(d.amount), 0);

      $('#metricTotalCount').text(totalCount);
      $('#metricShzpCount').text(shzpCount);
      $('#metricEdbCount').text(edbCount);
      $('#metricTotalAmount').text(`AED ${totalApprovedAmount.toLocaleString()}`);
    }
  </script>
