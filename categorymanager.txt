<script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.2/dist/js/bootstrap.bundle.min.js"></script>
<script>
  let TREE_DATA = [
    {
      id: 1,
      name: 'Electronics',
      children: [
        {
          id: 101,
          name: 'Laptops',
          children: [
            { id: 1001, name: 'Gaming Laptops' },
            { id: 1002, name: 'Ultrabooks' }
          ]
        },
        { id: 102, name: 'Smartphones' }
      ]
    },
    {
      id: 2,
      name: 'Clothing',
      children: [
        { id: 201, name: 'Men\'s Wear' },
        { id: 202, name: 'Women\'s Wear' }
      ]
    }
  ];

  let nodeModal;
  let jsonModal;

  document.addEventListener('DOMContentLoaded', () => {
    nodeModal = new bootstrap.Modal(document.getElementById('nodeModal'));
    jsonModal = new bootstrap.Modal(document.getElementById('jsonModal'));
    renderTree();
  });

  function renderTree() {
    document.getElementById('categoryTreeContainer').innerHTML = buildTreeHtml(TREE_DATA);
  }

  // Display JSON in Modal
  function showJsonModal() {
    const formattedJson = JSON.stringify(TREE_DATA, null, 2);
    document.getElementById('jsonOutput').textContent = formattedJson;
    jsonModal.show();
  }

  // Copy JSON to Clipboard
  function copyJsonToClipboard() {
    const jsonText = JSON.stringify(TREE_DATA, null, 2);
    navigator.clipboard.writeText(jsonText).then(() => {
      alert('JSON structure copied to clipboard!');
    }).catch(err => {
      console.error('Failed to copy: ', err);
    });
  }

  // Recursive Tree Builder with Add, Edit, and Delete action buttons
  function buildTreeHtml(nodes, isRoot = true) {
    let html = `<div class="${isRoot ? 'list-group list-group-flush' : 'tree-group collapse show'}">`;

    nodes.forEach(node => {
      const hasChildren = node.children && node.children.length > 0;
      const targetId = `collapse-node-${node.id}`;

      html += `
        <div class="list-group-item border-0 px-0 py-1">
          <div class="d-flex align-items-center justify-content-between">
            <div class="d-flex align-items-center">
              ${hasChildren ? `
                <button class="btn btn-sm btn-link text-dark p-0 me-1 tree-toggle" 
                        type="button" 
                        data-bs-toggle="collapse" 
                        data-bs-target="#${targetId}">
                  <i class="bi bi-chevron-down toggle-icon"></i>
                </button>
              ` : '<span class="tree-toggle d-inline-block"></span>'}
              
              <div class="form-check m-0">
                <input class="form-check-input node-checkbox" type="checkbox" value="${node.id}" id="chk-${node.id}">
                <label class="form-check-label ms-1" for="chk-${node.id}">${node.name}</label>
              </div>
            </div>

            <!-- Action Button Group -->
            <div class="btn-group btn-group-sm ms-2" role="group">
              <button class="btn btn-outline-primary py-0 px-1" title="Add Child" onclick="openNodeModal('add', ${node.id})">
                <i class="bi bi-plus-lg"></i>
              </button>
              <button class="btn btn-outline-secondary py-0 px-1" title="Edit Category" onclick="openNodeModal('edit', ${node.id})">
                <i class="bi bi-pencil"></i>
              </button>
              <button class="btn btn-outline-danger py-0 px-1" title="Delete Category" onclick="deleteNode(${node.id})">
                <i class="bi bi-trash"></i>
              </button>
            </div>
          </div>

          ${hasChildren ? buildTreeHtml(node.children, false) : `<div id="${targetId}" class="tree-group collapse show"></div>`}
        </div>
      `;
    });

    html += `</div>`;
    return html;
  }

  // Unified Modal Handler for Add & Edit
  function openNodeModal(mode, nodeId) {
    document.getElementById('modalMode').value = mode;
    document.getElementById('targetNodeId').value = nodeId || '';
    
    const nameInput = document.getElementById('nodeNameInput');
    const titleElem = document.getElementById('modalTitle');

    if (mode === 'add') {
      titleElem.innerText = nodeId ? 'Add Child Item' : 'Add Root Category';
      nameInput.value = '';
    } else if (mode === 'edit') {
      titleElem.innerText = 'Edit Category Name';
      const node = findNodeById(TREE_DATA, nodeId);
      nameInput.value = node ? node.name : '';
    }

    nodeModal.show();
  }

  // Save Function: Handles both Add and Edit actions
  function saveNode() {
    const name = document.getElementById('nodeNameInput').value.trim();
    const mode = document.getElementById('modalMode').value;
    const targetId = document.getElementById('targetNodeId').value;

    if (!name) return;

    if (mode === 'add') {
      const newNode = { id: Date.now(), name: name };
      if (!targetId) {
        TREE_DATA.push(newNode);
      } else {
        const parentNode = findNodeById(TREE_DATA, parseInt(targetId));
        if (parentNode) {
          if (!parentNode.children) parentNode.children = [];
          parentNode.children.push(newNode);
        }
      }
    } else if (mode === 'edit') {
      const nodeToEdit = findNodeById(TREE_DATA, parseInt(targetId));
      if (nodeToEdit) {
        nodeToEdit.name = name;
      }
    }

    nodeModal.hide();
    renderTree();
  }

  // Delete Function: Recursively removes a node and all sub-children
  function deleteNode(id) {
    const nodeToDelete = findNodeById(TREE_DATA, id);
    if (!nodeToDelete) return;

    if (confirm(`Are you sure you want to delete "${nodeToDelete.name}" and all its sub-categories?`)) {
      TREE_DATA = removeNodeById(TREE_DATA, id);
      renderTree();
    }
  }

  // Recursive Helper to Find Node
  function findNodeById(nodes, id) {
    for (let node of nodes) {
      if (node.id === id) return node;
      if (node.children) {
        const found = findNodeById(node.children, id);
        if (found) return found;
      }
    }
    return null;
  }

  // Recursive Helper to Remove Node
  function removeNodeById(nodes, id) {
    return nodes.filter(node => {
      if (node.id === id) return false;
      if (node.children) {
        node.children = removeNodeById(node.children, id);
      }
      return true;
    });
  }
</script>
